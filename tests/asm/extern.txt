// Métodos extern definidos en C (lib.c y libc). Devuelve 0 si todo está bien, o el número del chequeo que falló.
int neg() extern;
int suma3(int a, int b, int c) extern;
boolean esPar(int x) extern;
int abs(int x) extern;

int doble(int x) { return x * 2; }

int main() {
    // 1-3: un int negativo devuelto por C se ve negativo
    if (!(neg() < 0)) { return 1; }
    if (!(neg() == -1)) { return 2; }
    if (!(neg() + 1 == 0)) { return 3; }

    // 4: argumentos que son expresiones y llamadas
    if (!(suma3(doble(2), neg(), 1 + 1) == 5)) { return 4; }

    // 5-6: bool de C
    if (!esPar(4)) { return 5; }
    if (esPar(3)) { return 6; }

    // 7: función de la biblioteca estándar
    if (!(abs(-7) == 7)) { return 7; }
    return 0;
}
