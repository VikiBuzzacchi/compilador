// opt: deadcode
int f(int x) {
    if (true) {
        x = x + 1;
    } else {
        x = x - 1;
    }
    if (false) {
        x = 0;
    }
    while (false) {
        x = x * 2;
    }
    if (x > 2) {
        x = 3;
    }
    return x;
    x = 99;
}
int g(int n) {
    while (true) {
        if (n > 10) {
            return n;
        }
        n = n + 1;
    }
    return 0;
}
int main() {
    boolean b;
    b = false || f(1) > 2;
    return f(1) + g(5);
}
